STT2: LD A,(DE)
		CP 20H ;If there is a blank character after FD, load the rest as a file name (file name can be up to 32 characters)
		JR Z,SDLOAD
		CP '/' ;If FD is followed by '/', the following will not be loaded and executed as a file name (file name can be up to 32 characters)
		JR Z,SDLOAD
		CP 0DH ;If only FD has a new line, load DEFNAME string as file name
		JR NZ,STETC ;If not applicable, check other commands
STT3: PUSH DE ;Transfer configuration file name (0000.mzt)

STETC:

;****1BYTE reception ****
;Set received DATA to A register and return

		CALL F1CHK ;LOOP until PORTC BIT7 becomes 1
		IN A,(0D9h) ;PORTB -> A
		PUSH AF
		LD A,05H
		OUT (0DBH),A ;PORTC BIT2 <-1
CALL F2CHK ;LOOP until PORTC BIT7 becomes 0
		LD A,04H
		OUT (0DBH),A ;PORTC BIT2 <-0
		POP AF
		RET
		
;****1BYTE transmission ****
;Send the contents of the A register to the lower 4 BITs of PORTA, 4 bits at a time
SNDBYTE:
		PUSH AF
		RRA
		RRA
		RRA
		RRA
		AND 0FH
		CALL SND4BIT
		POP AF
		AND 0FH
		CALL SND4BIT
		RET

;****4BIT transmission ****
;Send lower 4 bits of A register
SND4BIT:
		OUT (0D8H),A
		LD A,05H
		OUT (0DBH),A ;PORTC BIT2 <-1
		CALL F1CHK ;LOOP until PORTC BIT7 becomes 1
		LD A,04H
		OUT (0DBH),A ;PORTC BIT2 <-0
CALL F2CHK
		RET
		
;****CHECK(1) BUSY ****
; 82H LOP until BIT7 becomes 1
F1CHK: IN A,(0DAH)
		AND 80H ;PORTC BIT7 = 1?
		JR Z,F1CHK
		RET

;****CHECK(0) BUSY ****
; LOOP until 82H BIT7 becomes 0
F2CHK: IN A,(0DAH)
		AND 80H ;PORTC BIT7 = 0?
		JR NZ,F2CHK
		RET

;******Get FILE NAME (IN:DE Next character after command character OUT:HL Start of file name)**********

STFN1: INC DE ; Skip spaces until file name
		LD A,(DE)
		CP 20H
		JR Z,STFN1
		CP 30H ;If the character is not ``0'' or more, it will be an error.
JP C,STSV2
		EX DE,HL
		POP AF
		RET

;****Send command (IN:A command code)****
STCD: CALL SNDBYTE ;Send command code of A register
		CALL RCVBYTE ;Get status (00H=OK)
		RET

;****Send file name (IN:HL start of file name) ******

		INC HL
		DEC B
		JR NZ,STFS1
		LD A,0DH
		CALL SNDBYTE
		CALL RCVBYTE ;Get status (00H=OK)
		RET

;****Command, file name transmission (IN:A command code HL: beginning of file name)****
STCMD: CALL STFN ;Get file name
		PUSH HL
CALL STCD ;Send command code
		POP HL
		AND A ;ERROR if other than 00
		JP NZ,SVER0
		CALL STFS ;Send file name
		AND A ;ERROR if other than 00
		JP NZ,SVER0
		RET
